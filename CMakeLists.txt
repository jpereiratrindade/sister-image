cmake_minimum_required(VERSION 3.20)
project(sister_image VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(TIFF 4.5 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(SISTER_IMAGE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(SISTER_IMAGE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(image_core src/RasterWindowAnalysis.cpp src/service.cpp
  src/NormalityModel.cpp src/RegimeStateMachine.cpp
  src/VectorShape.cpp src/RasterClip.cpp)
target_include_directories(image_core PUBLIC include)
target_include_directories(image_core SYSTEM PUBLIC vendor)
target_link_libraries(image_core PUBLIC TIFF::TIFF OpenSSL::Crypto Threads::Threads)
target_compile_options(image_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(sister-image-http src/main.cpp)
target_link_libraries(sister-image-http PRIVATE image_core)
target_compile_options(sister-image-http PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(image_core_test tests/core.cpp)
target_link_libraries(image_core_test PRIVATE image_core)
enable_testing()
add_test(NAME image_core COMMAND image_core_test)
add_test(NAME image_http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py $<TARGET_FILE:sister-image-http> ${CMAKE_CURRENT_SOURCE_DIR})
set_tests_properties(image_http PROPERTIES TIMEOUT 60)
add_executable(image_benchmark EXCLUDE_FROM_ALL tests/benchmark.cpp)
target_link_libraries(image_benchmark PRIVATE image_core)
add_test(NAME image_runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_test.py ${CMAKE_CURRENT_SOURCE_DIR})
set_tests_properties(image_runtime PROPERTIES TIMEOUT 60)
add_executable(image_geotiff_test tests/geotiff.cpp)
target_link_libraries(image_geotiff_test PRIVATE image_core)
add_test(NAME image_geotiff COMMAND image_geotiff_test)
